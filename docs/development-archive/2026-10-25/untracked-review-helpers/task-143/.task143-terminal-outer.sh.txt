restore() {
  rc=$?
  trap - EXIT
  cp .task143-terminal-disabled-source tixmoapi2/src/api/payments/service.ts
  cmp .task143-terminal-disabled-source tixmoapi2/src/api/payments/service.ts || exit 99
  echo "SOURCE_RESTORED status=$rc"
  exit "$rc"
}
trap restore EXIT
python3 - <<'ACTIVATE'
from pathlib import Path
p=Path('tixmoapi2/src/api/payments/service.ts');s=p.read_text()

assert s.count(a)==1
p.write_text(s.replace(a,'const EXPIRED_WEBHOOK_RECOVERY_ENABLED = true;'))
ACTIVATE
