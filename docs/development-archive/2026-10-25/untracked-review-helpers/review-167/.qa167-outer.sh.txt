restore() {
  rc=$?
  trap - EXIT
  cp .qa167-disabled-source tixmoapi2/src/api/payments/service.ts
  cmp .qa167-disabled-source tixmoapi2/src/api/payments/service.ts || exit 99
  timeout --signal=TERM --kill-after=5s 40s npm --prefix tixmoapi2 run build > .qa167-disabled-build.log 2>&1 || exit 98
  echo "SOURCE_RESTORED status=$rc DISABLED_BUILD=0"
  exit "$rc"
}
